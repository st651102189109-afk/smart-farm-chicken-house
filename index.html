// ================= FIREBASE ================= const firebaseConfig = { databaseURL: "YOUR_FIREBASE_URL" }; firebase.initializeApp(firebaseConfig); const db = firebase.database(); // ================= TAB ================= function showTab(id, btn) { document.querySelectorAll('.tab') .forEach(t => t.classList.remove('active')); document.querySelectorAll('.nav-btn') .forEach(b => b.classList.remove('active')); document.getElementById(id) .classList.add('active'); btn.classList.add('active'); } // ================= READ DATA ================= db.ref("/coop").on("value", (snapshot) => { const d = snapshot.val(); if (!d) return; // Sensor document.getElementById("temp").innerText = (d.temperature || 0) + "°C"; document.getElementById("humid").innerText = (d.humidity || 0) + "%"; document.getElementById("water").innerText = (d.waterLevel || 0) + "%"; document.getElementById("food").innerText = (d.foodLevel || 0) + "%"; // Status document.getElementById("dot") .classList.add("online"); document.getElementById("status-text") .innerText = "Online"; // Auto Config document.getElementById("auto-mode") .checked = d.mode === "AUTO"; document.getElementById("feed-time1") .value = d.feedTime1 || "06:00"; document.getElementById("feed-time2") .value = d.feedTime2 || "18:00"; document.getElementById("feed-duration") .value = d.feedDuration || 5; // Light Status updateLightUI(d.light); }); // ================= SAVE CONFIG ================= function saveConfig() { db.ref("/coop").update({ mode: document.getElementById("auto-mode").checked ? "AUTO" : "MANUAL", feedTime1: document.getElementById("feed-time1").value, feedTime2: document.getElementById("feed-time2").value, feedDuration: parseInt( document.getElementById("feed-duration").value ) || 5 }); alert("บันทึกสำเร็จ"); } // ================= LIGHT ================= function toggleLight() { db.ref("/coop/light") .once("value", (snap) => { let state = snap.val(); db.ref("/coop") .update({ light: state === "ON" ? "OFF" : "ON" }); }); } function updateLightUI(state) { const btn = document.getElementById("light-btn"); if (!btn) return; if (state === "ON") { btn.innerHTML = ' ปิดไฟ'; } else { btn.innerHTML = ' เปิดไฟ'; } } // ================= FEED ================= function feedNow() { db.ref("/coop") .update({ feedNow: true }); alert("สั่งให้อาหารแล้ว"); }
